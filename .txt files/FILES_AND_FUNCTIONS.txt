CLASS: User(Document)
    COLLECTION: 'users'
    DESCRIPTION: User accounts with authentication
    
    FIELDS:
        username          → StringField(required, unique) - Login username
        email             → StringField(unique) - Email address
        password          → StringField(required) - Hashed password (PBKDF2)
        role              → StringField(default='user') - 'user' or 'admin'
        reset_code        → StringField() - Password reset verification code
        reset_code_expiry → DateTimeField() - Reset code expiration time
    
    METHODS:
        set_password(plain_password)
            - Hashes password using Django's PBKDF2 (600k iterations)
            - Stores hashed password in password field
            - Returns: None
        
        check_password(plain_password)
            - Verifies password against stored hash
            - Returns: Boolean (True if correct, False if wrong)

CLASS: Favorite(Document)
    COLLECTION: 'favorites'
    DESCRIPTION: User favorite movies/shows
    
    FIELDS:
        username  → StringField(required) - User who favorited
        movie_id  → StringField(required) - ID of favorited movie
    
    METHODS: (Inherited from Document - save, delete, etc.)

    FIELDS:

        watching_percentage   → FloatField(default=0.0) - User-editable (0-100)

    METHODS: (Inherited from Document)

    ENDPOINT: POST /api/login/
    DESCRIPTION: JWT authentication endpoint
    
    METHODS:
        post(request)
            INPUT: {username: str, password: str}
            PROCESS:
                1. Validates username exists in MongoDB
                2. Checks password with User.check_password()
                3. Generates JWT access token (1 hour)
                4. Generates JWT refresh token (7 days)
            OUTPUT: {
                access: "jwt_token",
                refresh: "refresh_token",
                username: "john",
                role: "user"
            }
            STATUS: 200 OK, 400 Bad Request, 401 Unauthorized

CLASS: RecommendationEngine
    DESCRIPTION: AI-powered recommendation system
    TECHNOLOGY: TensorFlow Hub Universal Sentence Encoder v4
    
    INITIALIZATION:
        __init__(self)
            - Loads TensorFlow Hub model
            - Initializes SimpleVectorDB
            - Creates 'movies' and 'user_profiles' collections
    
    ═══════════════════════════════════════════════════════════════════════
    PART 1: EMBEDDING GENERATION
    ═══════════════════════════════════════════════════════════════════════
    
    METHOD: generate_embedding(text: str) -> List[float]
        DESCRIPTION: Converts text to 512-dimensional vector
        INPUT: Movie description text
        OUTPUT: [0.023, -0.145, 0.678, ..., 0.234] (512 numbers)
        USES: TensorFlow Hub Universal Sentence Encoder
    
    METHOD: create_movie_text(movie) -> str
        DESCRIPTION: Converts movie object to text description
        INPUT: Movie or TVShow document
        OUTPUT: "Primary Genre: Action. Type: Movie. Year: 2020. Plot: ..."
        LOGIC:
            - Emphasizes primary_genre (appears 3 times)
            - Includes: type, year, plot, rating, director
    
    ═══════════════════════════════════════════════════════════════════════
    PART 2: MOVIE INDEXING
    ═══════════════════════════════════════════════════════════════════════
    
    METHOD: index_movie(movie) -> bool
        DESCRIPTION: Indexes movie for AI recommendations
        PROCESS:
            1. Creates movie text with create_movie_text()
            2. Generates embedding with TensorFlow
            3. Stores in vector_db/movies/ collection
            4. Saves metadata (title, genres, year, etc.)
        RETURNS: True if successful, False if error
    
    METHOD: index_all_movies() -> Dict
        DESCRIPTION: Indexes all movies in database
        PROCESS:
            1. Gets all Movie + TVShow documents
            2. Indexes each with index_movie()
            3. Shows progress (e.g., "Indexed 100/500")
        RETURNS: {
            total: int,
            indexed: int,
            failed: int
        }
    
    METHOD: is_indexed(movie_id: str) -> bool
        DESCRIPTION: Check if movie already indexed
        RETURNS: Boolean

    
    ═══════════════════════════════════════════════════════════════════════
    PART 3: USER PROFILE CREATION
    ═══════════════════════════════════════════════════════════════════════
    
    METHOD: build_user_profile(username: str) -> Optional[List[float]]
        DESCRIPTION: Creates user preference embedding
        PROCESS:
            1. Calculates genre weights from:
               - Completed movies: 300 points each
               - Recent completed (7 days): 3x multiplier (900 pts)
               - Continue watching (>20%): 100 points
               - Favorites: 80 points
               - Manual genre selection: 150 points
            2. Creates weighted text:
               "Action Action Action Drama Drama Comedy"
               (repeated based on weights)
            3. Generates embedding from text
            4. Stores in vector_db/user_profiles/
        RETURNS: 512-dimensional user preference vector
    
    METHOD: _calculate_genre_weights(username: str) -> dict
        DESCRIPTION: Calculate genre importance scores
        RETURNS: {
            "Action": 3000,
            "Drama": 1800,
            "Comedy": 400
        }
    
    METHOD: get_top_user_genres(username: str) -> List[str]
        DESCRIPTION: Identify important genres (>= 15% of total weight)
        PROCESS:
            1. Calculates genre weights
            2. Total weight: 5200 points
            3. Threshold: 5200 * 0.15 = 780 points
            4. Returns genres above threshold
        RETURNS: ["Action", "Drama"] (genres with >= 15% importance)
        
        EXAMPLE:
            Action: 3000 pts (57%) → IMPORTANT
            Drama: 1800 pts (34%) → IMPORTANT
            Comedy: 400 pts (7%) → NOT important
    
    ═══════════════════════════════════════════════════════════════════════
    PART 4: RECOMMENDATION GENERATION (MULTI-GENRE FIX)
    ═══════════════════════════════════════════════════════════════════════
    
    METHOD: get_recommendations(username: str, count: int = 20, 
                                genre: str = None, type: str = None) -> List[Dict]
        DESCRIPTION: Generate personalized recommendations
        
        FIXED BUG: Multi-genre recommendations now properly interleaved
        
        PROCESS:
            1. Get user's important genres (>= 15% threshold)
            2. Calculate genre proportions:
               - Primary genre: 60% of results
               - Secondary genre: 30% of results
               - Others: 10% of results
            
            3. IF SINGLE GENRE:
               - Build single user profile embedding
               - Query vector DB once
               - Return top N similar movies
            
            4. IF MULTIPLE GENRES:
               - Query vector DB separately for EACH genre
               - Get proportional results per genre:
                 * Primary: 12 movies (60% of 20)
                 * Secondary: 6 movies (30% of 20)
                 * Others: 2 movies (10% of 20)
               - INTERLEAVE results (not cluster):
                 [A,A,D,A,D,A,A,D,A,D,A,A,D,O,A,D,A,A,O,D]
            
            5. Filter out:
               - Already watched (completed movies)
               - Disliked movies
               - Low recommendation_score movies (AI feedback)
            
            6. Apply additional filters (genre, type if provided)
        
        RETURNS: [{
            id: str,
            title: str,
            poster: str,
            genres: str,
            similarity: float,
            recommendation_reason: str
        }, ...]

    
    METHOD: get_similar_movies(movie_id: str, count: int = 10) -> List[Dict]

        PROCESS:
            1. Gets movie embedding from vector DB
            2. Queries for similar embeddings (cosine similarity)
            3. Returns top N most similar
        RETURNS: [{id, title, similarity, ...}, ...]
    
    METHOD: get_trending(count: int = 20) -> List[Dict]
        DESCRIPTION: Get trending/popular movies
        PROCESS:
            1. Queries recent movies (2020+)
            2. Sorts by: IMDb rating + watch count
            3. Returns top N
        RETURNS: [{id, title, trending_score, ...}, ...]

    DESCRIPTION: Check if user can access route
    PROCESS:
        1. Checks if user is logged in (token exists)
        2. If logged in → allow access (return true)
        3. If not logged in → redirect to /login
    USAGE: Applied to routes in app.routes.ts
    EXAMPLE:
        { path: 'dashboard', component: DashboardComponent, canActivate: [authGuard] }

│ DESCRIPTION: Automatically adds JWT token to all HTTP requests             │
└─────────────────────────────────────────────────────────────────────────────┘

FUNCTION: authInterceptor(req, next): Observable<HttpEvent<any>>
    DESCRIPTION: Intercepts all outgoing HTTP requests
    PROCESS:
        1. Gets access token from AuthService
        2. If token exists:
           - Clones request
           - Adds "Authorization: Bearer {token}" header
        3. Passes request to next handler
    USAGE: Configured in app.config.ts
    RESULT: Developer doesn't need to manually add auth header
